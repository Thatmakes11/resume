% ==================== 项目与工程经历 ====================
\section{项目与工程经历}

\noindent
\textbf{基于 GRPO 强化学习的多模态 GUI Agent 推理增强}
\begin{itemize}
    \item \textbf{数据构建与筛选:} 整合 OS-Atlas 等开源 GUI agent 数据集，收集横跨 Windows、Mac、Android 与 Web 的 14M Low Level 与 30K High Level 指令数据。通过 MLLM 实施 10 轮 Zero-Shot 启发式过滤，剔除极端难度样本并进行难度均衡采样，最终构建约 3K 高质量 RL 训练集
    \item \textbf{RLHF 训练:} 基于 GRPO 算法对 Qwen2.5-VL-3B-Instruct 进行强化学习微调，引导模型生成显式推理过程，并规范结构化输出（Action、Click Point、Input Text）。训练在 8xRTX4090 上完成，总训练时间约 2 天
    \item \textbf{奖励函数设计:} 设计任务执行驱动的复合奖励，包括：(1) 输出格式校验，(2) Action 类别匹配，(3) 空间坐标约束（点距或 Bounding Box），(4) 文本生成 F1 Score 评估，在 GUI benchmark 上的任务成功率较 SFT baseline 高 8\%
\end{itemize}

\vspace{0.3em}
\noindent
\textbf{多模态大模型 Prefill 阶段视觉 Token 动态压缩}
\begin{itemize}
    \item \textbf{视觉 Token 重要性建模:} 在指令序列尾部引入 Learnable Token，并在 Decoder 中后期提取其与视觉 Token 的注意力权重作为先验重要性分数
    \item \textbf{条件注意力重评分:} 结合视觉编码器提取的多尺度特征，在 $Q,K$ 矩阵中拼接视觉特征构建 Conditioned Self-Attention 模块，对先验重要性进行二次建模并计算最终 Token importance score
    \item \textbf{参数高效训练:} 冻结 Qwen2.5-VL-3B-Instruct 主干网络，仅训练新增 Token 与 Attention 层。利用 Bounding Box 区域 Mask 作为监督信号，并结合 DICE、BCE 与 L1 Loss 进行联合优化
    \item \textbf{推理优化:} 推理阶段采用 Top-K 策略保留 1/9 视觉 Token，实现约 9× Token 压缩。在 Visual CoT Benchmark 上仅使用 20K 样本微调（训练在 4×RTX4090 上完成，总训练时间约 2 小时）后性能损耗低于 10\%
\end{itemize}
